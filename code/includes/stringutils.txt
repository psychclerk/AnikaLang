def capitalize(s) {
    # First letter uppercase, rest lowercase
    sStr = str(s)
    if len(sStr) == 0 { return "" }
    return upper(left(sStr, 1)) + lower(mid(sStr, 1))
}

def titleCase(s) {
    # Capitalize first letter of each word
    sStr = str(s)
    words = split(sStr, " ")
    result = ""
    i = 0
    while i < len(words) {
        word = words[i]
        if len(word) > 0 {
            word = upper(left(word, 1)) + lower(mid(word, 1))
        }
        if i > 0 {
            result = result + " "
        }
        result = result + word
        i = i + 1
    }
    return result
}

def swapCase(s) {
    # Swap upper to lower and vice versa
    sStr = str(s)
    result = ""
    i = 0
    while i < len(sStr) {
        c = mid(sStr, i, 1)
        if c == upper(c) {
            result = result + lower(c)
        } else {
            result = result + upper(c)
        }
        i = i + 1
    }
    return result
}

def camelCase(s) {
    # Convert "hello world" or "hello-world" to "helloWorld"
    sStr = lower(str(s))
    sStr = replace(sStr, "-", " ")
    sStr = replace(sStr, "_", " ")
    words = split(sStr, " ")
    result = ""
    i = 0
    while i < len(words) {
        word = words[i]
        if len(word) > 0 {
            if i == 0 {
                result = result + lower(word)
            } else {
                result = result + upper(left(word, 1)) + lower(mid(word, 1))
            }
        }
        i = i + 1
    }
    return result
}

def snakeCase(s) {
    # Convert "Hello World" or "helloWorld" to "hello_world"
    # FIXED: Properly handles existing separators to avoid double underscores
    sStr = str(s)
    result = ""
    i = 0
    while i < len(sStr) {
        c = mid(sStr, i, 1)
        # Replace spaces and hyphens with underscore
        if c == " " or c == "-" {
            # Only add underscore if result doesn't already end with one
            if len(result) > 0 and right(result, 1) != "_" {
                result = result + "_"
            }
        } else {
            # Insert underscore before uppercase letters (camelCase handling)
            if c == upper(c) and lower(c) != upper(c) and len(result) > 0 {
                lastChar = right(result, 1)
                # Only add underscore if last char is lowercase letter
                if lastChar == lower(lastChar) and lastChar != "_" {
                    result = result + "_"
                }
            }
            result = result + lower(c)
        }
        i = i + 1
    }
    return result
}

def kebabCase(s) {
    # Convert "Hello World" to "hello-world"
    snake = snakeCase(s)
    return replace(snake, "_", "-")
}

def lTrim(s) {
    # Remove leading whitespace only
    sStr = str(s)
    i = 0
    while i < len(sStr) {
        c = mid(sStr, i, 1)
        if c != " " and c != "	" and c != "\n" and c != "\r" {
            return mid(sStr, i)
        }
        i = i + 1
    }
    return ""
}

def rTrim(s) {
    # Remove trailing whitespace only
    sStr = str(s)
    i = len(sStr) - 1
    while i >= 0 {
        c = mid(sStr, i, 1)
        if c != " " and c != "	" and c != "\n" and c != "\r" {
            return left(sStr, i + 1)
        }
        i = i - 1
    }
    return ""
}

def strip(s, chars) {
    # Remove specific characters from both ends
    sStr = str(s)
    charStr = str(chars)
    # Remove from start
    while len(sStr) > 0 and contains(charStr, left(sStr, 1)) {
        sStr = mid(sStr, 1)
    }
    # Remove from end
    while len(sStr) > 0 and contains(charStr, right(sStr, 1)) {
        sStr = left(sStr, len(sStr) - 1)
    }
    return sStr
}

def padLeft(s, width, padChar) {
    # Pad string on left to reach width
    sStr = str(s)
    pc = " "
    if padChar != null and len(str(padChar)) > 0 {
        pc = left(str(padChar), 1)
    }
    while len(sStr) < width {
        sStr = pc + sStr
    }
    return sStr
}

def padRight(s, width, padChar) {
    # Pad string on right to reach width
    sStr = str(s)
    pc = " "
    if padChar != null and len(str(padChar)) > 0 {
        pc = left(str(padChar), 1)
    }
    while len(sStr) < width {
        sStr = sStr + pc
    }
    return sStr
}

def center(s, width, padChar) {
    # Center string within width
    sStr = str(s)
    pc = " "
    if padChar != null and len(str(padChar)) > 0 {
        pc = left(str(padChar), 1)
    }
    totalPad = width - len(sStr)
    if totalPad <= 0 { return sStr }
    leftPad = int(totalPad / 2)
    rightPad = totalPad - leftPad
    result = ""
    i = 0
    while i < leftPad {
        result = result + pc
        i = i + 1
    }
    result = result + sStr
    i = 0
    while i < rightPad {
        result = result + pc
        i = i + 1
    }
    return result
}

def repeatStr(s, n) {
    # Repeat string n times (renamed to avoid conflict with any built-in)
    sStr = str(s)
    count = int(n)
    if count <= 0 { return "" }
    result = ""
    i = 0
    while i < count {
        result = result + sStr
        i = i + 1
    }
    return result
}

def wordCount(s) {
    # Count words in string
    sStr = trim(str(s))
    if len(sStr) == 0 { return 0 }
    words = split(sStr, " ")
    count = 0
    i = 0
    while i < len(words) {
        if len(trim(words[i])) > 0 {
            count = count + 1
        }
        i = i + 1
    }
    return count
}

def lastIndexOf(s, sub) {
    # Find last occurrence of substring (-1 if not found)
    sStr = str(s)
    search = str(sub)
    if len(search) == 0 { return -1 }
    lastPos = -1
    pos = index_of(sStr, search)
    offset = 0
    while pos >= 0 {
        lastPos = offset + pos
        offset = offset + pos + len(search)
        if offset >= len(sStr) {
            pos = -1
        } else {
            remaining = mid(sStr, offset)
            pos = index_of(remaining, search)
        }
    }
    return lastPos
}

def countOccurrences(s, sub) {
    # Count non-overlapping occurrences of substring
    sStr = str(s)
    search = str(sub)
    if len(search) == 0 { return 0 }
    count = 0
    offset = 0
    while offset < len(sStr) {
        remaining = mid(sStr, offset)
        pos = index_of(remaining, search)
        if pos < 0 {
            offset = len(sStr)
        } else {
            count = count + 1
            offset = offset + pos + len(search)
        }
    }
    return count
}

def extractBetween(s, startMarker, endMarker) {
    # Extract text between two markers
    sStr = str(s)
    startPos = index_of(sStr, startMarker)
    if startPos < 0 { return "" }
    afterStart = mid(sStr, startPos + len(startMarker))
    endPos = index_of(afterStart, endMarker)
    if endPos < 0 { return "" }
    return left(afterStart, endPos)
}

def before(s, marker) {
    # Get text before first occurrence of marker
    sStr = str(s)
    pos = index_of(sStr, marker)
    if pos < 0 { return sStr }
    return left(sStr, pos)
}

def after(s, marker) {
    # Get text after first occurrence of marker
    sStr = str(s)
    pos = index_of(sStr, marker)
    if pos < 0 { return "" }
    return mid(sStr, pos + len(marker))
}

def isEmpty(s) {
    # Check if string is empty or whitespace only
    if len(trim(str(s))) == 0 { return true }
    return false
}

def isNumeric(s) {
    # Check if string contains only digits (0-9)
    sStr = str(s)
    if len(sStr) == 0 { return false }
    i = 0
    while i < len(sStr) {
        c = mid(sStr, i, 1)
        if c != "0" and c != "1" and c != "2" and c != "3" and c != "4" and c != "5" and c != "6" and c != "7" and c != "8" and c != "9" {
            return false
        }
        i = i + 1
    }
    return true
}

def isAlpha(s) {
    # Check if string contains only letters
    sStr = str(s)
    if len(sStr) == 0 { return false }
    i = 0
    while i < len(sStr) {
        c = mid(sStr, i, 1)
        upperC = upper(c)
        lowerC = lower(c)
        # If upper == lower, it's not a letter (digit, symbol, space)
        if upperC == lowerC {
            return false
        }
        i = i + 1
    }
    return true
}

def isAlphanum(s) {
    # Check if string contains only letters and digits
    sStr = str(s)
    if len(sStr) == 0 { return false }
    i = 0
    while i < len(sStr) {
        c = mid(sStr, i, 1)
        isLetter = false
        isDigit = false
        if upper(c) != lower(c) {
            isLetter = true
        }
        if c == "0" or c == "1" or c == "2" or c == "3" or c == "4" or c == "5" or c == "6" or c == "7" or c == "8" or c == "9" {
            isDigit = true
        }
        if isLetter == false and isDigit == false {
            return false
        }
        i = i + 1
    }
    return true
}

def isEmail(s) {
    # Basic email validation
    sStr = trim(str(s))
    if len(sStr) == 0 { return false }
    atPos = index_of(sStr, "@")
    if atPos <= 0 { return false }
    dotPos = lastIndexOf(sStr, ".")
    if dotPos < atPos { return false }
    if dotPos == len(sStr) - 1 { return false }
    return true
}

def isUrl(s) {
    # Basic URL validation
    sStr = lower(trim(str(s)))
    if starts_with(sStr, "http://") { return true }
    if starts_with(sStr, "https://") { return true }
    if starts_with(sStr, "ftp://") { return true }
    return false
}

def isUpper(s) {
    # Check if all letters are uppercase
    sStr = str(s)
    if len(sStr) == 0 { return false }
    if sStr != upper(sStr) { return false }
    return true
}

def isLower(s) {
    # Check if all letters are lowercase
    sStr = str(s)
    if len(sStr) == 0 { return false }
    if sStr != lower(sStr) { return false }
    return true
}

def isPalindrome(s) {
    # Check if string reads same forwards and backwards (ignoring spaces and case)
    sStr = lower(str(s))
    sStr = replace(sStr, " ", "")
    rev = reverse(sStr)
    if sStr == rev { return true }
    return false
}

def reverse(s) {
    # Reverse a string
    sStr = str(s)
    result = ""
    i = len(sStr) - 1
    while i >= 0 {
        result = result + mid(sStr, i, 1)
        i = i - 1
    }
    return result
}

def rot13(s) {
    # ROT13 cipher (letters only)
    sStr = str(s)
    alpha = "abcdefghijklmnopqrstuvwxyz"
    result = ""
    i = 0
    while i < len(sStr) {
        c = mid(sStr, i, 1)
        lowerC = lower(c)
        pos = index_of(alpha, lowerC)
        if pos >= 0 {
            newPos = mod(pos + 13, 26)
            newC = mid(alpha, newPos, 1)
            if c == upper(c) {
                newC = upper(newC)
            }
            result = result + newC
        } else {
            result = result + c
        }
        i = i + 1
    }
    return result
}

def urlEncode(s) {
    # Percent-encode special characters for URLs
    sStr = str(s)
    result = ""
    i = 0
    while i < len(sStr) {
        c = mid(sStr, i, 1)
        isSafe = false
        if isAlphanum(c) {
            isSafe = true
        }
        if c == "-" or c == "_" or c == "." or c == "~" {
            isSafe = true
        }
        if isSafe == true {
            result = result + c
        } else {
            if c == " " {
                result = result + "%20"
            } else {
                code = charCode(c)
                hex = decToHex(code)
                if len(hex) < 2 {
                    hex = "0" + hex
                }
                result = result + "%" + upper(hex)
            }
        }
        i = i + 1
    }
    return result
}

def slugify(s) {
    # Convert to URL-friendly slug
    sStr = lower(trim(str(s)))
    sStr = replace(sStr, " ", "-")
    sStr = replace(sStr, "_", "-")
    result = ""
    i = 0
    while i < len(sStr) {
        c = mid(sStr, i, 1)
        if isAlphanum(c) or c == "-" {
            # Avoid consecutive dashes
            if c == "-" and len(result) > 0 and right(result, 1) == "-" {
                # Skip this dash
            } else {
                result = result + c
            }
        }
        i = i + 1
    }
    # Remove leading/trailing dashes
    result = trim(result)
    while len(result) > 0 and left(result, 1) == "-" {
        result = mid(result, 1)
    }
    while len(result) > 0 and right(result, 1) == "-" {
        result = left(result, len(result) - 1)
    }
    return result
}

def truncate(s, maxLen) {
    # Truncate string with default "..." suffix
    return truncateWithSuffix(s, maxLen, "...")
}

def truncateWithSuffix(s, maxLen, suffix) {
    # Truncate string with custom suffix
    sStr = str(s)
    suf = "..."
    if suffix != null and len(str(suffix)) > 0 {
        suf = str(suffix)
    }
    if len(sStr) <= maxLen { return sStr }
    cutLen = maxLen - len(suf)
    if cutLen < 0 { cutLen = 0 }
    return left(sStr, cutLen) + suf
}

def wrap(s, width) {
    # Word wrap text at specified width
    sStr = str(s)
    w = int(width)
    if w <= 0 { return sStr }
    words = split(sStr, " ")
    result = ""
    lineLen = 0
    i = 0
    while i < len(words) {
        word = words[i]
        if len(word) == 0 {
            i = i + 1
        } else {
            if lineLen == 0 {
                result = result + word
                lineLen = len(word)
            } else {
                if lineLen + 1 + len(word) > w {
                    result = result + "\n" + word
                    lineLen = len(word)
                } else {
                    result = result + " " + word
                    lineLen = lineLen + 1 + len(word)
                }
            }
            i = i + 1
        }
    }
    return result
}

def indent(s, nSpaces) {
    # Indent each line by n spaces
    sStr = str(s)
    lines = split(sStr, "\n")
    spaces = ""
    i = 0
    while i < nSpaces {
        spaces = spaces + " "
        i = i + 1
    }
    result = ""
    i = 0
    while i < len(lines) {
        if i > 0 {
            result = result + "\n"
        }
        result = result + spaces + lines[i]
        i = i + 1
    }
    return result
}

def join(list, delim) {
    # Join list elements into string with delimiter
    d = ""
    if delim != null { d = str(delim) }
    result = ""
    i = 0
    while i < len(list) {
        if i > 0 {
            result = result + d
        }
        result = result + str(list[i])
        i = i + 1
    }
    return result
}

def charAt(s, idx) {
    # Get character at index (0-based)
    sStr = str(s)
    i = int(idx)
    if i < 0 or i >= len(sStr) { return "" }
    return mid(sStr, i, 1)
}

def charCode(c) {
    # Get ASCII code of character
    # FIXED: Complete 95-character printable ASCII table (32-126)
    char = str(c)
    if len(char) == 0 { return 0 }
    ch = left(char, 1)
    # All 95 printable ASCII characters in order (space=32 to ~=126)
    chars = " !\"#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\]^_`abcdefghijklmnopqrstuvwxyz{|}~"
    pos = index_of(chars, ch)
    if pos >= 0 {
        return pos + 32
    }
    return 0
}

def fromCharCode(n) {
    # Get character from ASCII code
    # FIXED: Complete 95-character printable ASCII table (32-126)
    code = int(n)
    if code < 32 or code > 126 { return "" }
    # All 95 printable ASCII characters in order (space=32 to ~=126)
    chars = " !\"#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\]^_`abcdefghijklmnopqrstuvwxyz{|}~"
    idx = code - 32
    return mid(chars, idx, 1)
}

def firstChar(s) {
    # Get first character
    sStr = str(s)
    if len(sStr) == 0 { return "" }
    return left(sStr, 1)
}

def lastChar(s) {
    # Get last character
    sStr = str(s)
    if len(sStr) == 0 { return "" }
    return right(sStr, 1)
}

def numberFormat(n, decimals, thousandsSep, decimalSep) {
    # Format number with thousands separator
    num = float(n)
    dec = int(decimals)
    tSep = ","
    if thousandsSep != null and len(str(thousandsSep)) > 0 {
        tSep = left(str(thousandsSep), 1)
    }
    dSep = "."
    if decimalSep != null and len(str(decimalSep)) > 0 {
        dSep = left(str(decimalSep), 1)
    }
    rounded = round(num, dec)
    strNum = str(rounded)
    
    # Split into integer and decimal parts
    dotPos = index_of(strNum, ".")
    intPart = ""
    decPart = ""
    if dotPos >= 0 {
        intPart = left(strNum, dotPos)
        decPart = mid(strNum, dotPos + 1)
    } else {
        intPart = strNum
    }
    
    # Pad decimal part if needed
    while len(decPart) < dec {
        decPart = decPart + "0"
    }
    if dec > 0 {
        decPart = left(decPart, dec)
    }
    
    # Add thousands separators
    isNeg = false
    if left(intPart, 1) == "-" {
        isNeg = true
        intPart = mid(intPart, 1)
    }
    result = ""
    i = len(intPart) - 1
    count = 0
    while i >= 0 {
        if count > 0 and mod(count, 3) == 0 {
            result = tSep + result
        }
        result = mid(intPart, i, 1) + result
        count = count + 1
        i = i - 1
    }
    if isNeg == true {
        result = "-" + result
    }
    if dec > 0 {
        result = result + dSep + decPart
    }
    return result
}

def currencyFormat(n, symbol) {
    # Format as currency
    sym = "$"
    if symbol != null and len(str(symbol)) > 0 {
        sym = str(symbol)
    }
    return sym + numberFormat(n, 2, ",", ".")
}

def ordinal(n) {
    # Convert number to ordinal (1st, 2nd, 3rd, etc.)
    num = int(n)
    lastTwo = mod(num, 100)
    lastOne = mod(num, 10)
    suffix = "th"
    if lastTwo < 11 or lastTwo > 13 {
        if lastOne == 1 {
            suffix = "st"
        }
        if lastOne == 2 {
            suffix = "nd"
        }
        if lastOne == 3 {
            suffix = "rd"
        }
    }
    return str(num) + suffix
}

def lineCount(s) {
    # Count lines in string
    sStr = str(s)
    if len(sStr) == 0 { return 0 }
    count = 1
    i = 0
    while i < len(sStr) {
        if mid(sStr, i, 1) == "\n" {
            count = count + 1
        }
        i = i + 1
    }
    return count
}

def uniqueWords(s) {
    # Get list of unique words
    sStr = lower(trim(str(s)))
    words = split(sStr, " ")
    unique = []
    i = 0
    while i < len(words) {
        word = trim(words[i])
        if len(word) > 0 {
            found = false
            j = 0
            while j < len(unique) {
                if unique[j] == word {
                    found = true
                }
                j = j + 1
            }
            if found == false {
                list_append(unique, word)
            }
        }
        i = i + 1
    }
    return unique
}

def mostCommonWord(s) {
    # Find most frequently occurring word
    sStr = lower(trim(str(s)))
    words = split(sStr, " ")
    bestWord = ""
    bestCount = 0
    i = 0
    while i < len(words) {
        word = trim(words[i])
        if len(word) > 0 {
            count = 0
            j = 0
            while j < len(words) {
                if trim(words[j]) == word {
                    count = count + 1
                }
                j = j + 1
            }
            if count > bestCount {
                bestCount = count
                bestWord = word
            }
        }
        i = i + 1
    }
    return bestWord
}

def longestWord(s) {
    # Find longest word in string
    sStr = trim(str(s))
    words = split(sStr, " ")
    longest = ""
    i = 0
    while i < len(words) {
        word = trim(words[i])
        if len(word) > len(longest) {
            longest = word
        }
        i = i + 1
    }
    return longest
}

def equalsIgnoreCase(a, b) {
    # Case-insensitive comparison
    if lower(str(a)) == lower(str(b)) { return true }
    return false
}

def levenshtein(a, b) {
    # Edit distance between two strings
    # FIXED: Optimized to O(n²) using proper matrix updates
    s1 = str(a)
    s2 = str(b)
    len1 = len(s1)
    len2 = len(s2)
    
    # Optimization: if one string is empty, distance is length of other
    if len1 == 0 { return len2 }
    if len2 == 0 { return len1 }
    
    # Use two-row approach for memory efficiency
    prevRow = []
    currRow = []
    
    # Initialize previous row [0, 1, 2, ..., len2]
    j = 0
    while j <= len2 {
        list_append(prevRow, j)
        j = j + 1
    }
    
    # Fill in the rest row by row
    i = 1
    while i <= len1 {
        # Start current row with [i, 0, 0, ...]
        currRow = []
        list_append(currRow, i)
        j = 1
        while j <= len2 {
            c1 = mid(s1, i - 1, 1)
            c2 = mid(s2, j - 1, 1)
            cost = 0
            if c1 != c2 {
                cost = 1
            }
            # Calculate three possible operations
            delVal = prevRow[j] + 1
            insVal = currRow[j - 1] + 1
            subVal = prevRow[j - 1] + cost
            
            # Find minimum
            minVal = delVal
            if insVal < minVal { minVal = insVal }
            if subVal < minVal { minVal = subVal }
            
            list_append(currRow, minVal)
            j = j + 1
        }
        # Move current row to previous for next iteration
        prevRow = currRow
        i = i + 1
    }
    return prevRow[len2]
}

def template(s, dict) {
    # Replace {key} placeholders with dictionary values
    # FIXED: Uses dict_keys() for proper dict iteration
    sStr = str(s)
    result = sStr
    keys = dict_keys(dict)
    i = 0
    while i < len(keys) {
        key = keys[i]
        placeholder = "{" + key + "}"
        value = ""
        try {
            value = str(dict[key])
        } catch err {
            value = ""
        }
        result = replace(result, placeholder, value)
        i = i + 1
    }
    return result
}

def mask(s, visibleChars, maskChar) {
    # Mask string showing only last N characters
    sStr = str(s)
    visible = int(visibleChars)
    mc = "*"
    if maskChar != null and len(str(maskChar)) > 0 {
        mc = left(str(maskChar), 1)
    }
    if len(sStr) <= visible { return sStr }
    maskLen = len(sStr) - visible
    masked = ""
    i = 0
    while i < maskLen {
        masked = masked + mc
        i = i + 1
    }
    return masked + right(sStr, visible)
}
